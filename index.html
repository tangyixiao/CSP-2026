#include <bits/stdc++.h>
using namespace std;
const int N = 500000+5;
int n;
long long m;
long long dp[5005][5005];
long long a[N];
long long minn=0x3f3f3f3f;
long long ans=0;
int search(long long l,long long r)
{
	if(r-l==1)return l;
	long long mid=(l+r)/2;
	long long d=0;
	for(long long i=1;i<=n;i++)
	{
		if(a[i]<mid)d+=mid-a[i];
		else break;
		if(d>m)break;
	}
	if(d>m)search(l,mid);
	if(d<m)search(mid,r);
