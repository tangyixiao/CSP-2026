const int N = 500000+5;

long long m;

int search(long long l,long long r)
{
	if(r-l==1)return l;
	long long mid=(l+r)/2;
	long long d=0;
	for(long long i=1;i<=n;i++)
	{
		if(a[i]<mid)d+=mid-a[i];
		else break;
		if(d>m)break;
	}
	if(d>m)search(l,mid);
	if(d<m)search(mid,r);
